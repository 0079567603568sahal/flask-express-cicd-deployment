# Connect to your EC2 instance
ssh -i "mohdsahal924.pem" ubuntu@<EC2_PUBLIC_IP>

# Update and install dependencies
sudo apt update -y
sudo apt install -y python3 python3-pip git curl
curl -fsSL https://deb.nodesource.com/setup_18.x | sudo -E bash -
sudo apt install -y nodejs
sudo npm install -g pm2

# Clone your repos
cd ~
git clone https://github.com/0079567603568sahal/flask-backend.git
git clone https://github.com/0079567603568sahal/express-frontend.git

# Setup Flask
cd flask-backend
pip3 install -r requirements.txt
pm2 start "python3 app.py" --name flask-app

# Setup Express
cd ~/express-frontend
npm install
pm2 start "node index.js" --name express-app

# Save pm2 processes
pm2 save
pm2 list
